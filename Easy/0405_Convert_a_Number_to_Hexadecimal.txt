class Solution:
    def toHex(self, num: int) -> str:
        if num==0:
            return "0"
        if num<0:
            num=2**32+num
        result=""
        while num>0:
            remainder=num%16
            if remainder<10:
                result=chr(remainder+ord('0'))+result
            else:
                result=chr(remainder-10+ord('a'))+result
            num=num//16
        return result
