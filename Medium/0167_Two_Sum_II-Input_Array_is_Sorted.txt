class Solution:
    def twoSum(self, nums: list[int], target: int) -> list[int]:
        n=len(nums)
        left=0
        right=n-1
        while(left<right):
            sum=nums[left]+nums[right]
            if sum==target:
                return [left+1,right+1]
            elif sum>target:
                right-=1
            else:
                left+=1
        return []
