class Solution:
    def largestNumber(self, nums: List[int]) -> str:
        num=list(map(str,nums))
        for i in range(len(num)):
            for j in range(i+1,len(num)):
                if num[i]+num[j]<num[j]+num[i]:
                    num[i],num[j]=num[j],num[i]
        if num[0]=="0":
            return "0"
        return "".join(num)
