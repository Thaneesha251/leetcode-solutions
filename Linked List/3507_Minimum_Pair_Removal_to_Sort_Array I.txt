class Solution:
    def minimumPairRemoval(self, nums: List[int]) -> int:
        oper=0
        while any(nums[i]>nums[i+1] for i in range(len(nums)-1)):
            min_sum=float('inf')
            index=0
            for i in range(len(nums)-1):
                total=nums[i]+nums[i+1]
                if total<min_sum:
                    min_sum=total
                    index=i
            nums[index]=min_sum
            nums.pop(index+1)
            oper+=1
        return oper
