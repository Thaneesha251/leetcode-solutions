class MyHashMap:

    def __init__(self):
        self.keys=[]
        self.values=[]

    def put(self,key,value):
        if key in self.keys:
            index=self.keys.index(key)
            self.values[index]=value
        else:
            self.keys.append(key)
            self.values.append(value)

    def get(self,key):
        if key in self.keys:
            index=self.keys.index(key)
            return self.values[index]
        return -1

    def remove(self,key):
        if key in self.keys:
            index=self.keys.index(key)
            self.keys.pop(index)
            self.values.pop(index)
